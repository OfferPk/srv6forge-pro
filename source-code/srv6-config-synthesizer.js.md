  function synthesizeCiscoIOSXR(cfg) {  
    var loc \= cfg.locatorBlock || 'fcbb:bb00::/32';  
    var locName \= cfg.locatorName || 'LOC01';  
    var asn \= cfg.bgpAsn || '65000';  
    var vrf \= cfg.tenantVrf || 'VRF\_PROD';  
    var isMicroSid \= \!\!cfg.isMicroSid;  
    var tilfa \= \!\!cfg.enableTiLfa;  
    var sbfd \= \!\!cfg.enableSbfd;  
    var encap \= cfg.encapsulationMode || 'H.Encaps.Red';

    var lines \= \[  
      '\! \====================================================================',  
      '\! Cisco IOS-XR (v7.3+) SRv6 / uSID Network Programming Configuration',  
      '\! Generated by SRv6Forge Pro \- RFC 8754 / RFC 8986 Compliant',  
      '\! \====================================================================',  
      'router isis 100',  
      ' is-type level-2-only',  
      ' net 49.0001.0000.0000.0001.00',  
      ' address-family ipv6 unicast',  
      '  metric-style wide',  
      '  segment-routing srv6',  
      '   locator ' \+ locName,  
      '  \!'  
    \];

    if (tilfa) {  
      lines.push('  fast-reroute per-prefix');  
      lines.push('  fast-reroute per-prefix ti-lfa');  
    }  
    lines.push(' \!');  
    lines.push(' interface TenGigE0/0/0/0');  
    lines.push('  point-to-point');  
    lines.push('  address-family ipv6 unicast');  
    lines.push(' \!');  
    lines.push('\!');  
    lines.push('segment-routing srv6');  
    lines.push(' encapsulation');  
    lines.push('  source-address ' \+ (cfg.sourceIp || '2001:db8:ffff::1'));  
    if (encap \=== 'H.Encaps.Red') {  
      lines.push('  reduced-srh');  
    }  
    lines.push(' \!');  
    lines.push(' locators');  
    lines.push('  locator ' \+ locName);  
    if (isMicroSid) {  
      lines.push('   micro-segment behavior uncompressed');  
      lines.push('   prefix ' \+ loc \+ ' block-len 32 node-len 16 func-bits 16');  
    } else {  
      lines.push('   prefix ' \+ loc);  
    }  
    lines.push('  \!');  
    lines.push(' \!');  
    lines.push('\!');  
    lines.push('router bgp ' \+ asn);  
    lines.push(' bgp router-id ' \+ (cfg.routerId || '192.0.2.1'));  
    lines.push(' address-family vpnv4 unicast');  
    lines.push('  segment-routing srv6');  
    lines.push('   locator ' \+ locName);  
    lines.push('   alloc mode per-vrf');

    lines.push(' \!');

    lines.push('   segment-routing srv6');  
    lines.push('    alloc mode per-vrf');

    lines.push('  \!');  
    lines.push(' \!');  
    lines.push('\!');

      lines.push('sbfd');  
      lines.push(' local-discriminator 10001');

    }

  }

  function synthesizeJuniperJunos(cfg) {  
    var loc \= cfg.locatorBlock || 'fcbb:bb00::/32';  
    var locName \= cfg.locatorName || 'LOC01';  
    var asn \= cfg.bgpAsn || '65000';  
    var vrf \= cfg.tenantVrf || 'VRF\_PROD';  
    var tilfa \= \!\!cfg.enableTiLfa;

    var lines \= \[  
      '\# \====================================================================',  
      '\# Juniper Junos (21.4R1+) SRv6 Configuration Hierarchy',  
      '\# Generated by SRv6Forge Pro \- RFC 8754 / RFC 8986 Compliant',  
      '\# \====================================================================',  
      'set routing-options source-packet-routing srv6 locator ' \+ locName \+ ' ' \+ loc,  
      'set routing-options source-packet-routing srv6 encapsulation source-address ' \+ (cfg.sourceIp || '2001:db8:ffff::1'),  
      'set protocols isis source-packet-routing srv6 locator ' \+ locName,  
      'set protocols isis level 2 wide-metrics-only'  
    \];

    if (tilfa) {  
      lines.push('set protocols isis backup-coverage level 2 post-convergence-lfa srv6');  
    }

    lines.push('set protocols bgp group SRV6\_PEERS type internal');  
    lines.push('set protocols bgp group SRV6\_PEERS family inet-vpn unicast');  
    lines.push('set protocols bgp group SRV6\_PEERS family inet6-vpn unicast');  
    lines.push('set routing-instances ' \+ vrf \+ ' instance-type vrf');  
    lines.push('set routing-instances ' \+ vrf \+ ' route-distinguisher 192.0.2.1:100');  
    lines.push('set routing-instances ' \+ vrf \+ ' vrf-target target:' \+ asn \+ ':100');  
    lines.push('set routing-instances ' \+ vrf \+ ' protocols srv6 locator ' \+ locName);  
    lines.push('set routing-instances ' \+ vrf \+ ' protocols srv6 end-dt4');

    return lines.join('\\n');  
  }

      'sysctl \-w net.ipv6.conf.eth0.seg6\_enabled=1',  
      '',  
      '\# 2\. Add SRv6 Traffic Engineering / Encap Route via iproute2',  
      'ip route add 10.10.10.0/24 encap seg6 mode ' \+ encap \+ ' segs ' \+ sidStr \+ ' dev eth0',
